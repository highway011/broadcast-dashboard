const weather = document.getElementById("weather"); const crew = document.getElementById("crew"); const firstWeatherTime = 120000; // 2 minutes const normalTime = 60000; // 1 minute let showingWeather = true; let firstWeather = true; function rotate() { if (showingWeather) { // Weather -> Crew weather.classList.remove("active"); crew.classList.add("active"); showingWeather = false; setTimeout(rotate, normalTime); } else { // Crew -> Weather crew.classList.remove("active"); weather.classList.add("active"); showingWeather = true; if (firstWeather) { firstWeather = false; setTimeout(rotate, firstWeatherTime); } else { setTimeout(rotate, normalTime); } } } // Initial weather page stays up for 2 minutes setTimeout(rotate, firstWeatherTime);
